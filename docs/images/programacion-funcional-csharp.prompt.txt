Bottom large code card "Ejemplo completo · Program.cs":
using System;
using System.Linq;
int[] numeros = { -1, 2, 3 };
Func<int, int> duplicar = n => n * 2;
int[] resultado = numeros
    .Where(n => n > 0)
    .Select(duplicar)
    .ToArray();
Console.WriteLine(string.Join(", ", resultado));
